using System;

class EmployeeSalaryCalculator
{
    static void Main()
    {
        Console.WriteLine("=== Employee Salary Calculator ===");

        // Input
        Console.Write("Enter Employee Name: ");
        string name = Console.ReadLine();
	
        Console.Write("Enter Basic Salary: ");
        double basicSalary = Convert.ToDouble(Console.ReadLine());

        Console.Write("Enter Overtime Hours: ");
        double overtimeHours = Convert.ToDouble(Console.ReadLine());

        Console.Write("Enter Overtime Rate per Hour: ");
        double overtimeRate = Convert.ToDouble(Console.ReadLine());

        Console.Write("Enter Allowance: ");
        double allowance = Convert.ToDouble(Console.ReadLine());

        Console.Write("Enter Tax Rate (%): ");
        double taxRate = Convert.ToDouble(Console.ReadLine());

        // Gross Salary calculation
        double grossSalary = basicSalary + (overtimeHours * overtimeRate) + allowance;

        // Tax calculation
        double tax = grossSalary * (taxRate / 100);

        // Net Salary calculation
        double netSalary = grossSalary - tax;

        // Boolean check: earns at least 50,000?
        bool earnsAtLeast50k = netSalary >= 50000;

        // Output
        Console.WriteLine("\n--- Salary Details ---");
        Console.WriteLine($"Employee: {name}");
        Console.WriteLine($"Basic Salary: {basicSalary:C}");
        Console.WriteLine($"Overtime Pay: {(overtimeHours * overtimeRate):C}");
        Console.WriteLine($"Allowance: {allowance:C}");
        Console.WriteLine($"Gross Salary: {grossSalary:C}");
        Console.WriteLine($"Tax ({taxRate}%): {tax:C}");
        Console.WriteLine($"Net Salary: {netSalary:C}");
        Console.WriteLine($"Earns at least 50,000? {earnsAtLeast50k}");
    }
}
